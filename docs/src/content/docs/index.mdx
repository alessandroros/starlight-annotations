  <Card title="Notes in your repo" icon="seti:git">

  </Card>
  <Card title="Anchored to the text" icon="pencil">

  </Card>
  <Card title="Threads" icon="comment">

  </Card>
  <Card title="Runs anywhere" icon="rocket">
